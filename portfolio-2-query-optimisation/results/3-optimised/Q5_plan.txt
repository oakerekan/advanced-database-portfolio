-- Q5: Cashier end-of-day: takings by payment method for one day
-- phase: 3-optimised

SELECT method, count(*) AS payments, sum(amount) AS total
FROM payment
WHERE paid_at >= TIMESTAMPTZ '2026-06-01 00:00+01'
  AND paid_at <  TIMESTAMPTZ '2026-06-02 00:00+01'
GROUP BY method
ORDER BY total DESC;
